export async function parallel<T>(values:Iterable<T>,concurrency:number,signal:AbortSignal,work:(value:T)=>Promise<void>):Promise<void> {
  const iterator=values[Symbol.iterator]();let firstError:unknown;
  const workers=Array.from({length:concurrency},async()=>{
    while(!signal.aborted&&firstError===undefined) {
      const next=iterator.next();if(next.done)return;
      try {await work(next.value);}catch(error){firstError??=error;}
    }
  });
  await Promise.all(workers);
  if(firstError!==undefined)throw firstError;
  signal.throwIfAborted();
}

  const data=value as Record<string,unknown>;
