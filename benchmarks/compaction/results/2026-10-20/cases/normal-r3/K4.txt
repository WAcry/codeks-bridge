import { setTimeout } from 'node:timers/promises';

export interface RetryOptions {
  retries:number;
  signal:AbortSignal;
  now?:()=>number;
  random?:()=>number;
  sleep?:(ms:number,signal:AbortSignal)=>Promise<void>;
  onRetry?:(attempt:number,delayMs:number)=>void;
}

export function retryDelay(header:string|null,now:number):number|undefined {
  if(!header)return undefined;
  const seconds=Number(header);
  if(Number.isFinite(seconds)&&seconds>=0)return seconds*1000;
  const date=Date.parse(header);
  if(!Number.isNaN(date))return Math.max(0,date-now);
  return undefined;
}

export async function retry<T>(
  operation:(attempt:number)=>Promise<T>,
  classify:(e:unknown)=>{retry:boolean;after?:string},
  options:RetryOptions
):Promise<T> {
  const now=options.now??Date.now, random=options.random??Math.random;
  const sleep=options.sleep??((ms,signal)=>setTimeout(ms,undefined,{signal}));
  for(let attempt=0;;attempt++) {
    options.signal.throwIfAborted();
    try {return await operation(attempt);}
    catch(error) {
      const decision=classify(error);
      if(!decision.retry||attempt>=options.retries)throw error;
      const after=retryDelay(decision.after??null,now());
      const base=Math.min(30000,500*2**attempt);
      const delay=after??Math.floor(base*(0.5+random()*0.5));
      options.onRetry?.(attempt+1,delay);
      await sleep(delay,options.signal);
    }
  }
}

Accumulated product requirements

CLI:
- `driftship push <directory> --to <url>`
- Default four workers, 4 MiB chunks, four extra retries after initial attempt (five total).
- Concurrency input range 1–16.
- Effective concurrency = min(requested, server `maximum_parallel_parts`); requested 2 remains 2.
- Global pool across files with bounded lazy scheduling.
- Non-dry-run authenticated discovery is mandatory.
- Malformed discovery or protocol other than 3 fails before begin; no guessed defaults.
- Known server constraints: min 1 MiB, max 16 MiB, preferred 4 MiB, max parallel parts 4, SHA-256.
- Empty files begin and complete with zero parts.

Resume and durability:
- Default `--resume=on`; support `--resume=off`.
- Reject `--resume=verify` for now.
- Reject old `--no-resume`, no alias; branch is unreleased.
- Journal inside source at `.driftship/resume-v1.json`.
- Resume off starts fresh locally only; never delete or abort remote sessions.
- Durable journal: temp file mode 0600, parent directory mode 0700, fsync file, rename, fsync directory.
- Serialize mutations with an in-process queue, no database or dependency.
- Journal write failure stops new scheduling and surfaces error.
- Preserve corrupt/newer journal rather than silently overwriting it.
- Identity uses full SHA-256, size, nanosecond mtime; full hashing cost accepted.
- Validate remote status size/hash/part size before accepting resumed data.
- Only missing/expired remote sessions restart automatically.
- Session mismatch is an error, not silent session replacement.
- No remote cleanup, aborts, or deletion. Earlier cleanup ideas were explicitly withdrawn.

Retries and reconciliation:
- Only idempotent PUT parts may retry.
- Retry connection loss and statuses 408, 429, 500, 502, 503, 504, honoring valid Retry-After.
- Never automatically retry begin or complete.
- Part may persist before connection loss; replay exact bytes safely.
- Complete may succeed before its response is lost; one GET status may reconcile.
- If that GET fails, return its error and let next invocation resume.
- Avoid duplicate objects or skipped parts.
- Uploaded bytes count confirmed unique bytes, not retransmission traffic.
