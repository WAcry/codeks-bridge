User requirements and decisions

Upload protocol and scheduling:
- Default workers: 4.
- Chunk size: 4 MiB, matching server.
- Retry count 4 means four retries after the initial attempt, up to five total attempts.
- CLI concurrency range: 1–16.
- Effective concurrency = minimum of requested concurrency and server `maximum_parallel_parts`; requesting 2 must stay 2.
- Authenticated discovery must validate protocol 3.
- Known server constraints: minimum part size 1 MiB, maximum 16 MiB, preferred 4 MiB, maximum parallel parts 4, SHA-256 required.
- Malformed capabilities or protocol other than 3 fail before begin. Do not guess defaults.
- Only idempotent PUT parts may retry.
- Retry transport failures and appropriate statuses (408, 429, 500, 502, 503, 504), respecting valid Retry-After seconds or HTTP dates.
- Never automatically retry begin or complete.
- After an ambiguous complete transport failure, exactly one status GET is allowed.
- If that GET fails, return its own error; next invocation resumes.
- Empty files begin and complete with no parts.
- Lost part response may occur after server persistence; retry must not skip parts or count duplicates.
- Lost complete response may occur after successful completion; must not create duplicate objects.

```ts
import { setTimeout } from 'node:timers/promises';

export interface RetryOptions {
  retries: number;
  signal: AbortSignal;
  now?: () => number;
  random?: () => number;
  sleep?: (ms: number, signal: AbortSignal) => Promise<void>;
  onRetry?: (attempt: number, delayMs: number) => void;
}

export function retryDelay(header: string | null, now: number): number | undefined {
  if (!header) return undefined;
  const seconds = Number(header);
  if (Number.isFinite(seconds) && seconds >= 0) return seconds * 1000;
  const date = Date.parse(header);
  if (!Number.isNaN(date)) return Math.max(0, date - now);
  return undefined;
}

export async function retry<T>(
  operation: (attempt: number) => Promise<T>,
  classify: (e: unknown) => { retry: boolean; after?: string },
  options: RetryOptions
): Promise<T> {
  const now = options.now ?? Date.now;
  const random = options.random ?? Math.random;
  const sleep = options.sleep ?? ((ms, signal) => setTimeout(ms, undefined, { signal }));
  for (let attempt = 0; ; attempt++) {
    options.signal.throwIfAborted();
    try {
      return await operation(attempt);
    } catch (error) {
      const decision = classify(error);
      if (!decision.retry || attempt >= options.retries) throw error;
      const after = retryDelay(decision.after ?? null, now());
      const base = Math.min(30000, 500 * 2 ** attempt);
      const delay = after ?? Math.floor(base * (0.5 + random() * 0.5));
      options.onRetry?.(attempt + 1, delay);
      await sleep(delay, options.signal);
    }
  }
}
```

export function validateSession(status: Record<string, unknown>, file: JournalFile): void {
  if (
    status.file_size !== file.size ||
    status.file_sha256 !== file.sha256 ||
    status.part_size !== file.chunkBytes
  ) {
    throw new CliError('Remote session does not match source', 'session_mismatch');
  }
}

export function* pendingParts(plans: Array<{ file: JournalFile; absolutePath: string }>) {
  for (const plan of plans) {
    const completed = new Set(plan.file.parts.map(part => part.number));
    for (let number = 1; number <= Math.ceil(plan.file.size / plan.file.chunkBytes); number++) {
      if (!completed.has(number)) yield { plan, number };
    }
  }
}
